  sxj-text : Text = "\"\\\n" & char-to-text (code-to-char 0) & "café À Ω 中"

  opening : [Console] Nothing =
    let a = config-add-model config-new sxj-text sxj-text
    in let b = config-add-sampler a sxj-text
    in let c = config-add-lora b sxj-text
    in let d = config-add-upscaler c sxj-text
    in let cfg = config-set-current-model d sxj-text
    in let state = ExplorerState { config = cfg, images-served = 0, prompts-seen = 0 }
    in act
      error <- respond-error sxj-text
      empty <- respond-json (jstr "")
      config <- dispatch state "GET /api/config"
      status <- dispatch state "GET /api/status"
      cache <- dispatch state "GET /api/cache-key?model_short=A\\B&sampler=euler&prompt_hash=p\"\\"
      plain <- respond-json (jstr "plain/ascii")
      print-line-uni "sd-json-end"
    end
